### 2. 멀티탭 실시간 상태 동기화 (`src/utils/broadcastSync.js`, `src/hooks/useStorageSync.js`)
- 사용자가 여러 브라우저 탭을 띄워두고 작업할 때, 한 탭에서 추가한 북마크/보관함/테마 변경이 다른 탭에도 새로고침 없이 즉시 반영되도록 `BroadcastChannel` API 구축:
  ```javascript
  const channel = typeof window !== 'undefined' && 'BroadcastChannel' in window
    ? new BroadcastChannel('redue_ai_sync_channel')
    : null;

  export function broadcastDataChange(type, payload) {
    if (channel) {
      channel.postMessage({ type, payload, timestamp: Date.now() });
    }
  }

  export function subscribeDataSync(callback) {
    if (!channel) return () => {};
    const handler = (event) => callback(event.data);
    channel.addEventListener('message', handler);
    return () => channel.removeEventListener('message', handler);
  }
useStorage.js 및 Bookmarks, SavedPhrases 추가/삭제 액션에 broadcastDataChange 연동.

### 2. WebRTC DataChannel 기반 오프라인 1:1 P2P 고속 전송기 (`src/utils/p2pTransfer.js`, `src/components/modals/P2PTransferModal.jsx`)
- **개념**: 해외 현지에서 클라우드 서버나 모바일 데이터 소진 없이 브라우저 간 직접 연결(P2P)로 고화질 사진, 음성 일기, 일정표를 초고속 전송:
  ```javascript
  export class P2PConnector {
    constructor(onMessageCallback) {
      this.peer = new RTCPeerConnection({
        iceServers: [{ urls: 'stun:stun.l.google.com:19302' }]
      });
      this.channel = null;
      this.onMessage = onMessageCallback;
    }

    createOffer() {
      return new Promise(async (resolve) => {
        this.channel = this.peer.createDataChannel('redue_p2p');
        this.channel.onmessage = (e) => this.onMessage(e.data);
        const offer = await this.peer.createOffer();
        await this.peer.setLocalDescription(offer);
        resolve(JSON.stringify(offer));
      });
    }

    async receiveAnswer(answerJson) {
      const answer = JSON.parse(answerJson);
      await this.peer.setRemoteDescription(new RTCSessionDescription(answer));
    }

    sendData(data) {
      if (this.channel && this.channel.readyState === 'open') {
        this.channel.send(data);
      }
    }
  }

- **주요 기능**:
  ```javascript
  // src/hooks/useDeviceOrientation.js
  import { useState, useEffect } from 'react';

  export function useDeviceOrientation() {
    const [heading, setHeading] = useState(0);
    const [isSupported, setIsSupported] = useState(false);

    useEffect(() => {
      const handleOrientation = (event) => {
        let compassHeading = event.alpha;
        if (event.webkitCompassHeading) {
          // iOS Safari 지원
          compassHeading = event.webkitCompassHeading;
        }
        if (compassHeading !== null) {
          setHeading(Math.round(compassHeading));
          setIsSupported(true);
        }
      };

      if (window.DeviceOrientationEvent) {
        window.addEventListener('deviceorientation', handleOrientation, true);
      }
      return () => window.removeEventListener('deviceorientation', handleOrientation);
    }, []);

    return { heading, isSupported };
  }

### 1. 오프라인 시맨틱 자연어 검색 엔진 (`src/utils/localSemanticSearch.js`)
- **개념**: 외부 AI 서버 통신 없이도 "조용하고 힐링되는 바다", "비 올 때 걷기 좋은 곳", "초보자 코스" 같은 사용자 자연어 질의를 분석하여 `src/data/` 내부의 랜드마크·순례길·휴양지 데이터를 코사인 유사도(Cosine Similarity) 및 가중치 키워드 벡터 매칭으로 실시간 추천:
  ```javascript
  // 텍스트 토큰화 및 가중치 추출 (초경량 TF-IDF / N-gram)
  function extractKeywords(text) {
    if (!text) return [];
    return text
      .toLowerCase()
      .replace(/[^\w\s가-힣]/g, ' ')
      .split(/\s+/)
      .filter((word) => word.length >= 2);
  }

  // 코사인 유사도 연산
  export function searchTravelDataOffline(query, datasets = []) {
    const queryTokens = extractKeywords(query);
    if (queryTokens.length === 0) return datasets;

    const scored = datasets.map((item) => {
      const targetText = `${item.title || ''} ${item.description || ''} ${item.tags?.join(' ') || ''} ${item.location || ''}`;
      const targetTokens = extractKeywords(targetText);

      let matches = 0;
      queryTokens.forEach((qt) => {
        if (targetTokens.some((tt) => tt.includes(qt) || qt.includes(tt))) {
          matches += 1;
        }
      });

      const score = matches / Math.sqrt(queryTokens.length * (targetTokens.length || 1));
      return { item, score };
    });

    return scored
      .filter((res) => res.score > 0)
      .sort((a, b) => b.score - a.score)
      .map((res) => res.item);
  }

### 3. 발음 정확도 채점 및 텍스트 유사도 알고리즘 (`src/utils/pronunciationScorer.js`)
- 브라우저 STT로 인식된 텍스트와 원문 텍스트 간의 일치도를 Levenshtein Distance(편집 거리) 기반으로 계산하여 0~100점 점수화:
  ```javascript
  export function calculatePronunciationScore(originalText, spokenText) {
    if (!spokenText) return { score: 0, feedback: '음성이 감지되지 않았습니다.' };

    const cleanOriginal = originalText.toLowerCase().replace(/[^\w\s가-힣]/g, '').trim();
    const cleanSpoken = spokenText.toLowerCase().replace(/[^\w\s가-힣]/g, '').trim();

    if (cleanOriginal === cleanSpoken) {
      return { score: 100, feedback: '완벽한 발음입니다! 현지인과 자연스럽게 소통할 수 있습니다.', matchedTokens: [] };
    }

    const origTokens = cleanOriginal.split(/\s+/);
    const spokenTokens = cleanSpoken.split(/\s+/);

    let matchCount = 0;
    const tokenStatus = origTokens.map(token => {
      const isMatch = spokenTokens.includes(token);
      if (isMatch) matchCount++;
      return { word: token, isMatch };
    });

    const baseScore = Math.round((matchCount / origTokens.length) * 100);
    const finalScore = Math.max(20, Math.min(100, baseScore));

    let feedback = '조금 더 자신감 있게 큰 목소리로 따라 읽어보세요.';
    if (finalScore >= 85) feedback = '원어민이 정확하게 알아들을 수 있는 훌륭한 발음입니다!';
    else if (finalScore >= 65) feedback = '의미 전달에 문제없는 양호한 발음입니다.';

    return {
      score: finalScore,
      feedback,
      tokenStatus
    };
  }

### 3. 일자별 맞춤 거리 자동 배분 알고리즘 (`src/utils/trailPaceCalculator.js`)
- 출발 지점과 도착 지점 사이의 실제 거점 데이터를 기반으로 무리하지 않는 숙박 거점을 자동 매칭하는 로직 작성:
  ```javascript
  export function calculateCustomTrailPlan({ waypoints, startId, endId, targetDays, dailyKm = 20 }) {
    const startIndex = waypoints.findIndex(w => w.id === startId);
    const endIndex = waypoints.findIndex(w => w.id === endId);
    if (startIndex === -1 || endIndex === -1 || startIndex >= endIndex) return null;

    const selectedWaypoints = waypoints.slice(startIndex, endIndex + 1);
    const totalDistance = selectedWaypoints.reduce((acc, cur, idx) => {
      return idx > 0 ? acc + (cur.distanceFromPrev || 0) : acc;
    }, 0);

    const planDays = targetDays || Math.ceil(totalDistance / dailyKm);
    const avgDailyDistance = totalDistance / planDays;

    // 일자별 최적 숙박 거점 분할 매핑
    const dailyStages = [];
    let currentAccum = 0;
    let dayCount = 1;
    let dayStartWaypoint = selectedWaypoints[0];

    for (let i = 1; i < selectedWaypoints.length; i++) {
      const wp = selectedWaypoints[i];
      currentAccum += (wp.distanceFromPrev || 0);

      // 목표 일일 거리에 도달했거나 마지막 거점인 경우 하루 일정 마감
      if (currentAccum >= avgDailyDistance || i === selectedWaypoints.length - 1) {
        dailyStages.push({
          day: dayCount,
          from: dayStartWaypoint.name,
          to: wp.name,
          distance: parseFloat(currentAccum.toFixed(1)),
          estimatedHours: (currentAccum / 4).toFixed(1), // 시속 4km 기준
          lodgingCount: wp.albergueCount || 3
        });
        dayStartWaypoint = wp;
        currentAccum = 0;
        dayCount++;
      }
    }

    return {
      totalDistance: parseFloat(totalDistance.toFixed(1)),
      totalDays: dailyStages.length,
      stages: dailyStages
    };
  }

닫기(X) 버튼 클릭 또는 배경 터치 시 소리가 중단되면서 부드럽게 원래 상세 모달로 복귀하는지 검수.
